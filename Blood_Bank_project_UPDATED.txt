CREATE database BBMS;

use BBMS;

create table LOGIN(
login_ID varchar(45) not null,
user_password varchar(50) not null,
login_username varchar(50) not null,
login_work_title_ID varchar(10),
primary key (login_ID)
);


create table WORK_TITLE(
WRK_TITLE_ID varchar(20) not null,
WRK_TITLE_NAME varchar(50) not null,
WRK_TITLE_DISCRIPTION varchar(50),
primary key (WRK_TITLE_ID)
);


CREATE TABLE USER1(
USER_ID varchar(45) NOT NULL,
USER_NAME varchar(45) NOT NULL,
STREET varchar(40),
CITY varchar(20),
PROVINCE varchar(40),
login_ID varchar(45) not null,
WRK_TITLE_ID varchar(20) not null,
primary key (USER_ID),
constraint fk_login foreign key(login_ID) references LOGIN(login_ID),
constraint fk_WRK_TITLE foreign key(WRK_TITLE_ID) references WORK_TITLE(WRK_TITLE_ID)
);


create table USER_PHONE(
USER_ID varchar(45) not null,
USER_MOBILE varchar(15) ,
primary key (USER_ID,USER_MOBILE),
constraint fk_USER foreign key(USER_ID) references USER1(USER_ID)
);



alter table LOGIN
add constraint fk_L_WRK_TITLE foreign key(login_work_title_ID) references WORK_TITLE(WRK_TITLE_ID);

create table PATIENT(
PAT_ID varchar(20) not null,
PAT_NAME varchar(50) NOT NULL,
PAT_GENDER varchar(10) NOT NULL,
PAT_BLOOD_GROUP varchar(10) NOT NULL,
PAT_STREET varchar(40),
PAT_CITY varchar(20),
PAT_PROVINCE varchar(40),
USER_ID varchar(45) NOT NULL,
GUARDIAN_ID varchar(20),
primary key(PAT_ID),
constraint fk_PAT_USER foreign key(USER_ID) references USER1(USER_ID),
constraint fk_P_GUARD foreign key(GUARDIAN_ID) references PATIENT(PAT_ID)
);

create table PATIENT_MOBILE(
PAT_ID varchar(20) NOT NULL,
PAT_MOBILE varchar(15),
primary key(PAT_ID,PAT_MOBILE),
constraint fk_P_MOBILE foreign key(PAT_ID) references PATIENT(PAT_ID)
);

create table DONER(
DONER_ID varchar(20) NOT NULL,
DONER_NAME varchar(50) NOT NULL,
DONER_AGE INT,
DONER_GENDER varchar(10) NOT NULL,
DONER_BLOOD_GROUP varchar(10) NOT NULL,
BLOOD_QUANTITY varchar(10) NOT NULL, -- Quality is changed
DONATION_DATE date,
EXPIRY_DATE date,
DONER_STREET varchar(40),
DONER_CITY varchar(20),
DONER_PROVINCE varchar(40),
USER_ID varchar(45) NOT NULL,
-- NEED TO ADD DRIVE ATTRIBUTE
primary key(DONER_ID),
constraint FK_DONER_USER foreign key(USER_ID) references USER1(USER_ID)
);

create table DONER_MOBILE(
DONER_ID varchar(20) NOT NULL,
DONER_MOBILE varchar(15) ,
primary key(DONER_ID,DONER_MOBILE),
constraint fk_D_MOBILE foreign key(DONER_ID) references DONER(DONER_ID)
);

create table DONER_DISEASE(
DONER_ID varchar(20) NOT NULL,
DONER_DISEASE varchar(15) NOT NULL,
primary key(DONER_ID,DONER_DISEASE),
constraint fk_D_DISEASE foreign key(DONER_ID) references DONER(DONER_ID)
);

create table ANALYST(
ANALYST_ID varchar(20) NOT NULL,
ANALYST_NAME varchar(50) NOT NULL,
primary key(ANALYST_ID)
);

create table BLOOD_BANK(
B_BANK_ID varchar(20) NOT NULL,
B_BANK_QUANTITY varchar(10),
BANK_STREET varchar(40),
BANK_CITY varchar(20),
BANK_PROVINCE varchar(40),
primary key(B_BANK_ID)
);

create table AVAILABLE_BLOOD_GROUP(
B_BANK_ID varchar(20) NOT NULL,
AVAILABLE_BLOOD_GROUP varchar(10),
primary key(B_BANK_ID, AVAILABLE_BLOOD_GROUP),
constraint FK_A_BLOOD_GROUP foreign key(B_BANK_ID) references BLOOD_BANK(B_BANK_ID)


);


create table BLOOD(
BLOOD_ID varchar(20) NOT NULL,
BLOOD_GROUP varchar(10),
BLOOD_DISCRIPTION varchar(40),
USER_ID varchar(45) NOT NULL,
DONER_ID varchar(20) NOT NULL,
ANALYST_ID varchar(20) NOT NULL,
B_BANK_ID varchar(20) NOT NULL,
primary key(BLOOD_ID),
constraint FK_B_USER foreign key(USER_ID) references USER1(USER_ID),
constraint FK_B_DONER foreign key(DONER_ID) references DONER(DONER_ID),
constraint FK_B_ANALYST foreign key(ANALYST_ID) references ANALYST(ANALYST_ID),
constraint FK_B_BANK foreign key(B_BANK_ID) references BLOOD_BANK(B_BANK_ID)
);
 


create table HOSPITAL(
HOSPITAL_ID varchar(20) NOT NULL,
HOSPITAL_NAME varchar(40),
H_STREET varchar(40),
H_CITY  varchar(20),
H_PROVINCE varchar(40),
primary key (HOSPITAL_ID)

);

create table HOSPITAL_MOBILE (
HOSPITAL_ID varchar(20) NOT NULL,
HOSPITAL_CONTACT_NO varchar(15),
primary key(HOSPITAL_ID,HOSPITAL_CONTACT_NO),
constraint FK_H_MOBILE foreign key(HOSPITAL_ID) references HOSPITAL(HOSPITAL_ID)

);

-- N TO N RELATIONSHIP TABLE

create table HOSPITAL_TO_B_BANK(
HOSPITAL_ID varchar(20),
B_BANK_ID varchar(20),
primary key(HOSPITAL_ID,B_BANK_ID),
constraint FK_C_HOSPITAL foreign key(HOSPITAL_ID) references HOSPITAL(HOSPITAL_ID),
constraint FK_C_BANK foreign key(B_BANK_ID) references BLOOD_BANK(B_BANK_ID)
);

create table BLOOD_DONATION_CAMP(
CAMP_NAME varchar(40) NOT NULL,
CAMP_DATE date,
B_BANK_ID varchar(20),
primary key(B_BANK_ID,CAMP_NAME),
constraint FK_ORGANIZE foreign key(B_BANK_ID) references BLOOD_BANK(B_BANK_ID)
);

create table BLOOD_COMPONENTS(
COMPONENT_TYPE varchar(20),
BLOOD_ID varchar(20),
primary key(BLOOD_ID,COMPONENT_TYPE ),
constraint FK_CONTAINS foreign key(BLOOD_ID) references BLOOD(BLOOD_ID)
);

CREATE VIEW PatientView AS
SELECT PAT_ID, PAT_NAME, PAT_BLOOD_GROUP, PAT_CITY FROM PATIENT;

CREATE VIEW DonerView AS
SELECT DONER_ID, DONER_NAME, DONER_BLOOD_GROUP, DONER_CITY FROM DONER;

    p.PAT_NAME AS Patient_Name,
    d.DONER_NAME AS Doner_Name,
    p.PAT_BLOOD_GROUP AS Patient_Blood_Group,
    d.DONER_BLOOD_GROUP AS Doner_Blood_Group,
    p.PAT_CITY AS City
FROM PatientView AS p
INNER JOIN DonerView AS d ON p.PAT_BLOOD_GROUP = d.DONER_BLOOD_GROUP AND p.PAT_CITY = d.DONER_CITY;

-- Nested_query1
-- The Nested_query1 to find the average age of male donors (DONER_GENDER = 'M')
-- who have donated blood to blood banks with a 'HIGH' quantity of blood (B_BANK_QUANTITY = 'HIGH').

SELECT AVG(DONER_AGE) AS AVG_MALE_DONOR_AGE
FROM DONER
WHERE DONER_ID IN (
    SELECT DONER_ID FROM BLOOD NATURAL JOIN BLOOD_BANK
    where B_BANK_QUANTITY = 'HIGH'
)
AND DONER_GENDER = 'M';

-- This query finds analysts who have analyzed blood samples from hospitals located in Colombo.
SELECT DISTINCT ANALYST_NAME
FROM ANALYST
WHERE ANALYST_ID IN (
    SELECT ANALYST_ID
    FROM BLOOD
    WHERE B_BANK_ID IN (
        SELECT B_BANK_ID
        FROM BLOOD_BANK
        WHERE BANK_CITY = 'Colombo'
    )
);
